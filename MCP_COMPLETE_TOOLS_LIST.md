## 🧪 PowerShell Test Script

```powershell
# Get JWT token
$loginBody = @{
    usernameOrEmail = "testuser"
    password = "Test@123"
} | ConvertTo-Json

$loginResponse = Invoke-WebRequest -Uri "http://localhost:9091/api/auth/login" `
    -Method POST `
    -Body $loginBody `
    -ContentType "application/json" `
    -UseBasicParsing

$token = ($loginResponse.Content | ConvertFrom-Json).jwtToken

# Test 1: Get all MCP tools
$headers = @{ "Authorization" = "Bearer $token" }
$toolsResponse = Invoke-WebRequest -Uri "http://localhost:9090/api/mcp/tools" `
    -Method GET `
    -Headers $headers `
    -UseBasicParsing

$tools = ($toolsResponse.Content | ConvertFrom-Json)
Write-Host "Total MCP Tools: $($tools.totalTools)" -ForegroundColor Green

# Display tool names by category
$tools.tools | Group-Object { $_.name.Split('_')[0] } | ForEach-Object {
    Write-Host "`n$($_.Name) Tools ($($_.Count)):" -ForegroundColor Cyan
    $_.Group | ForEach-Object {
        Write-Host "  - $($_.name)" -ForegroundColor White
    }
}

# Test 2: Get MCP status
$statusResponse = Invoke-WebRequest -Uri "http://localhost:9090/api/mcp/tools-status" `
    -Method GET `
    -Headers $headers `
    -UseBasicParsing

$status = ($statusResponse.Content | ConvertFrom-Json)
Write-Host "`nMCP Status:" -ForegroundColor Yellow
Write-Host "  Connected Clients: $($status.mcpClients)" -ForegroundColor White
Write-Host "  Status: $($status.status)" -ForegroundColor White
```
